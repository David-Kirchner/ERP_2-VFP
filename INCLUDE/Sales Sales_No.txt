Save_PrintedSOinSales_No()
PARAMETERS lPrinted, nSalesNum, nConn
	cSQL = "UPDATE dbo.Sales_No "
	IF lPrinted
		cSQL = cSQL + " SET PrintedSO = 1"
		cSQL = cSQL + ",[Changing]=0"
		cSQL = cSQL + ",[ShopRecv]=0"
	ELSE
		cSQL = cSQL + " SET PrintedSO = 0"
		cSQL = cSQL + ",[ShopRecv]=0"
	ENDIF
	cSQL = cSQL + " WHERE SalesNum = "+STR(nSalesNum)
